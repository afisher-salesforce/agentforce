// Allowlist — only these email domains can access content
const ALLOWED_DOMAINS = ['salesforce.com', 'blackbaud.com'];
// Shown briefly when a signed-in user's domain isn't allowed,
// then Clerk signs them out automatically
function DomainRejected() {
  const { signOut } = useClerk();
  useEffect(() => {
    signOut({ redirectUrl: basePath || '/' });
  }, [signOut]);
  return (
    <div className="flex min-h-[100dvh] items-center justify-center bg-[#f4f6f9] px-6">
      <div className="max-w-md w-full text-center">
        <div className="bg-white rounded-2xl p-8 border border-[#d6dde8] shadow-sm">
          <div className="text-3xl mb-4">🔒</div>
          <h2 className="text-xl font-bold text-[#1f2937] mb-3">Access Restricted</h2>
          <p className="text-[#5b6473] text-[15px] leading-relaxed">
            This site is available to <strong>salesforce.com</strong> and{' '}
            <strong>blackbaud.com</strong> email addresses only.
            You are being signed out.
          </p>
        </div>
      </div>
    </div>
  );
}
function MainApp() {
  const { isSignedIn, isLoaded, user } = useUser();
  // ... loading and signed-out states handled first ...
  // Domain check — runs after every sign-in
  const email  = user?.primaryEmailAddress?.emailAddress ?? '';
  const domain = email.split('@')[1]?.toLowerCase() ?? '';
  if (!ALLOWED_DOMAINS.includes(domain)) {
    return <DomainRejected />;   // renders briefly, then signOut() fires
  }
  // Allowed domain — render the full app
  return ( /* ... routes and Layout ... */ );
}

How it works in plain terms:

Clerk handles the actual sign-in/sign-up — any email provider works at that stage
Immediately after authentication, MainApp reads the user's primary email address from Clerk
It extracts the domain (user@salesforce.com → salesforce.com) and checks it against ALLOWED_DOMAINS
If the domain isn't on the list, DomainRejected renders — which simultaneously shows a message and triggers signOut() via useEffect
The user lands back on the public landing page, locked out
To add more domains in the future, just extend the array at the top:

const ALLOWED_DOMAINS = ['salesforce.com', 'blackbaud.com', 'newdomain.com'];
